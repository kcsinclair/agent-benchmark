// cache.mjs
export class LruTtlCache {
  constructor({ capacity, defaultTtlMs = Infinity, now = () => Date.now() }) {
    if (!Number.isInteger(capacity) || capacity < 1) {
      throw new RangeError('capacity must be an integer >= 1');
    }
    this._capacity = capacity;
    this._defaultTtlMs = defaultTtlMs;
    this._now = now;
    // key -> { value, expiry }
    this._map = new Map();
  }

  set(key, value, ttlMs) {
    const now = this._now();
    const expiry = ttlMs === undefined
      ? Infinity
      : ttlMs <= 0
        ? now // already expired, but we store it; it will be removed on access
        : now + ttlMs;

    // If key exists, update value and expiry, move to front (most recent)
    if (this._map.has(key)) {
      const entry = this._map.get(key);
      entry.value = value;
      entry.expiry = expiry;
      // Move to end (most recently used)
      this._map.delete(key);
      this._map.set(key, entry);
      return this;
    }

    // New entry
    const entry = { value, expiry };
    this._map.set(key, entry);

    // Evict if over capacity
    this._trim();

    return this;
  }

  get(key) {
    if (!this._map.has(key)) return undefined;

    const entry = this._map.get(key);
    const now = this._now();

    // Check expiry
    if (entry.expiry !== Infinity && now >= entry.expiry) {
      this._map.delete(key);
      return undefined;
    }

    // Move to most recently used (end of Map)
    const value = entry.value;
    this._map.delete(key);
    this._map.set(key, entry);
    return value;
  }

  has(key) {
    if (!this._map.has(key)) return false;

    const entry = this._map.get(key);
    const now = this._now();

    if (entry.expiry !== Infinity && now >= entry.expiry) {
      this._map.delete(key);
      return false;
    }

    return true;
  }

  delete(key) {
    if (!this._map.has(key)) return false;
    this._map.delete(key);
    return true;
  }

  get size() {
    const live = new Map();
    const now = this._now();

    for (const [key, entry] of this._map) {
      if (entry.expiry === Infinity || now < entry.expiry) {
        live.set(key, entry);
      }
      // expired entries are simply not counted
    }

    // Purge expired entries from the real map
    for (const key of this._map.keys()) {
      const entry = this._map.get(key);
      if (entry.expiry !== Infinity && now >= entry.expiry) {
        this._map.delete(key);
      }
    }

    return live.size;
  }

  keys() {
    // Build live ordered list from most recent to least recent
    // Map insertion order is LRU order (oldest -> newest).
    // We iterate reverse to get most recent first.
    const result = [];
    const now = this._now();

    // Iterate reverse to get MRU first, and purge expired on the fly
    const keys = Array.from(this._map.keys()).reverse();
    for (const key of keys) {
      const entry = this._map.get(key);
      if (!entry) continue; // already deleted
      if (entry.expiry === Infinity || now < entry.expiry) {
        result.unshift(key); // prepend maintains MRU order
      } else {
        // expired, remove from map
        this._map.delete(key);
      }
    }

    return result;
  }

  // Internal: ensure size <= capacity by evicting expired first, then LRU
  _trim() {
    // First pass: remove all expired entries
    const now = this._now();
    for (const [key, entry] of this._map) {
      if (entry.expiry !== Infinity && now >= entry.expiry) {
        this._map.delete(key);
      }
    }

    // Second pass: if still over capacity, evict LRU (oldest) entries
    while (this._map.size > this._capacity) {
      const firstKey = this._map.keys().next().value;
      if (firstKey === undefined) break;
      this._map.delete(firstKey);
    }
  }
}
