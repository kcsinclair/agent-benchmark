```python
import re

def parse_duration(text: str) -> float:
    """
    Parses a human-readable duration string and returns the total seconds as a float.
    """
    if not isinstance(text, str):
        raise TypeError("Input must be a string")

    text = text.strip()
    if not text:
        raise ValueError("Duration string cannot be empty")

    # Handle leading sign
    multiplier = 1.0
    if text.startswith('-'):
        multiplier = -1.0
        text = text[1:]
    elif text.startswith('+'):
        # Rule 1: No leading '+'
        raise ValueError("Leading '+' is not allowed")

    # Regex to find components: (number)(unit)
    # Number: digits, optional dot, digits. 
    # Must not start with dot, must not end with dot.
    # Unit: h, m, s, or ms.
    pattern = re.compile(r'^(\d+(?:\.\d+)?)(ms|h|m|s)$')
    
    # We will iterate through the string and extract components
    total_seconds = 0.0
    last_unit_rank = 5  # Rank: h=4, m=3, s=2, ms=1. Start higher than any.
    
    unit_map = {
        'h': (3600.0, 4),
        'm': (60.0, 3),
        's': (1.0, 2),
        'ms': (0.001, 1)
    }

    remaining = text
    if not remaining:
        raise ValueError("No duration components found")

    while remaining:
        # Try to find the longest possible match at the start of the string
        # Since units can be 's' or 'ms', we check for the match.
        # We use a regex that captures the number and the unit.
        match = re.match(r'^(\d+(?:\.\d+)?)(ms|h|m|s)', remaining)
        if not match:
            raise ValueError(f"Invalid component or format at: {remaining}")
        
        num_str, unit_str = match.groups()
        
        # Validate number format (regex handles most, but ensure no trailing dot)
        # The regex \d+(?:\.\d+)? ensures it starts with digit and doesn't end with dot.
        val = float(num_str)
        seconds_per_unit, rank = unit_map[unit_str]
        
        # Rule 2: Strictly descending order and no repeats
        if rank >= last_unit_rank:
            raise ValueError(f"Units must be in strictly descending order: {unit_str}")
        
        total_seconds += val * seconds_per_unit
        last_unit_rank = rank
        
        # Advance the string
        remaining = remaining[match.end():]

    return total_seconds * multiplier
```
