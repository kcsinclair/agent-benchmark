```python
import re

def parse_duration(text: str) -> float:
    """
    Parses a human-readable duration string and returns the total seconds as a float.
    """
    if not isinstance(text, str):
        raise TypeError("Input must be a string")

    text = text.strip()
    if not text:
        raise ValueError("Duration string cannot be empty")

    # Handle the leading sign
    is_negative = False
    if text.startswith('-'):
        is_negative = True
        text = text[1:]
    elif text.startswith('+'):
        # Rule 1: No leading '+'
        raise ValueError("Leading '+' is not allowed")

    # Regex to find components: (number)(unit)
    # Number: digits, optional dot, digits. 
    # No leading dot (.5), no trailing dot (5.), no exponents.
    # Unit: h, m, s, or ms.
    pattern = re.compile(r'^(\d+(?:\.\d+)?)(h|m|s|ms)$')
    
    # We will iterate through the string, extracting components one by one
    total_seconds = 0.0
    remaining = text
    
    # Define the required order and their multipliers
    units_info = [
        ('h', 3600.0),
        ('m', 60.0),
        ('s', 1.0),
        ('ms', 0.001),
    ]
    
    last_unit_index = -1
    components_found = 0

    while remaining:
        # Try to match the start of the string with a component
        # We use a regex that matches the start of the string
        match = re.match(r'^(\d+(?:\.\d+)?)(h|m|s|ms)', remaining)
        if not match:
            raise ValueError(f"Invalid duration format at: {remaining}")
        
        val_str, unit = match.groups()
        
        # Validate the number format specifically for trailing dots (regex \d+(?:\.\d+)? handles this)
        # but we must ensure the match consumed the whole number part correctly.
        # The regex above already prevents "5." because it requires digits after the dot.
        
        # Check unit order and uniqueness
        try:
            current_unit_index = units_info.index(unit)
        except ValueError:
            raise ValueError(f"Unknown unit: {unit}")
            
        if current_unit_index <= last_unit_index:
            raise ValueError(f"Units must be in strictly descending order and not repeated: {unit}")
        
        # Calculate value
        total_seconds += float(val_str) * units_info[current_unit_index][1]
        
        # Update state
        last_unit_index = current_unit_index
        components_found += 1
        remaining = remaining[match.end():]
        
        # Rule 5: Whitespace anywhere else is invalid
        if remaining.startswith(' '):
            raise ValueError("Whitespace is not allowed between components")

    if components_found == 0:
        raise ValueError("At least one component is required")

    return -total_seconds if is_negative else total_seconds
```
