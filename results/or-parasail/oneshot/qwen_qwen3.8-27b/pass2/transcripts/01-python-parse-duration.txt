import re


def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError(f"Expected str, got {type(text).__name__}")
    
    text = text.strip()
    
    if not text:
        raise ValueError("Empty duration string")
    
    # Check for leading minus
    negative = False
    if text.startswith('-'):
        negative = True
        text = text[1:]
    
    if not text:
        raise ValueError("No components after minus sign")
    
    # Check for any whitespace in the remaining text
    if any(c.isspace() for c in text):
        raise ValueError("Whitespace not allowed in duration string")
    
    # Define the pattern for a single component: number followed by unit
    # Number: digits with optional fractional part (no leading +, no exponents, no bare . or .5, no trailing dot)
    # Valid numbers: 2, 1.5, 0.25, 10, 0, etc.
    # Invalid: .5, 5., +5, 1e5, etc.
    
    # Unit: h, m, s, ms
    # We need to match components in order: h, m, s, ms
    
    # Let's use a regex to match the entire string as a sequence of components
    # Each component: (\d+(?:\.\d+)?)(h|m|s|ms)
    # But we need to ensure units are in strictly descending order and each appears at most once
    
    # Let's parse component by component
    # First, let's find all components using a regex that matches number+unit
    # But we need to be careful: "ms" should be matched as a unit, not "m" followed by "s"
    
    # Let's use a regex that matches the entire string
    # The pattern for a valid number: \d+(?:\.\d+)?
    # The pattern for a unit: h|m|s|ms
    
    # We'll use a regex to find all components, but we need to ensure the whole string is consumed
    # Let's try to match the entire string with a pattern that captures components
    
    # Actually, let's parse it manually by scanning through the string
    # We'll look for numbers followed by units
    
    # Let's use a regex to find all valid components in order
    # The regex for a component: (\d+(?:\.\d+)?)(h|ms|m|s)
    # Note: we put ms before m to ensure ms is matched correctly
    
    component_pattern = re.compile(r'(\d+(?:\.\d+)?)(h|ms|m|s)')
    
    # Find all matches
    matches = list(component_pattern.finditer(text))
    
    # Check that the matches cover the entire string
    if not matches:
        raise ValueError("No valid components found")
    
    # Check that the matches are contiguous and cover the entire string
    pos = 0
    for match in matches:
        if match.start() != pos:
            raise ValueError(f"Invalid character at position {pos}")
        pos = match.end()
    
    if pos != len(text):
        raise ValueError(f"Invalid character at position {pos}")
    
    # Now validate the order of units
    # Units must be in strictly descending order: h, m, s, ms
    # Each unit may appear at most once
    
    unit_order = {'h': 0, 'm': 1, 's': 2, 'ms': 3}
    
    seen_units = set()
    last_order = -1
    
    total_seconds = 0.0
    
    for match in matches:
        number_str = match.group(1)
        unit = match.group(2)
        
        if unit in seen_units:
            raise ValueError(f"Repeated unit: {unit}")
        
        current_order = unit_order[unit]
        if current_order <= last_order:
            raise ValueError(f"Units not in descending order: {unit} after previous unit")
        
        seen_units.add(unit)
        last_order = current_order
        
        number = float(number_str)
        
        if unit == 'h':
            total_seconds += number * 3600
        elif unit == 'm':
            total_seconds += number * 60
        elif unit == 's':
            total_seconds += number
        elif unit == 'ms':
            total_seconds += number / 1000.0
    
    if negative:
        total_seconds = -total_seconds
    
    return total_seconds
