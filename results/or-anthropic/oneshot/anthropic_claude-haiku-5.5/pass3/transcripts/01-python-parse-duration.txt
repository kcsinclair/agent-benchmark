```python
import re

_COMPONENT = r"(\d+(?:\.\d+)?)(ms|h|m|s)"
_UNIT_ORDER = {"h": 0, "m": 1, "s": 2, "ms": 3}
_TOKEN_RE = re.compile(_COMPONENT)
_FULL_RE = re.compile(rf"(?:{_COMPONENT})+")

_FACTORS = {
    "h": 3600.0,
    "m": 60.0,
    "s": 1.0,
    "ms": 0.001,
}


def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError(f"parse_duration expects str, got {type(text).__name__}")

    s = text.strip()
    if not s:
        raise ValueError("empty duration string")

    sign = 1.0
    if s.startswith("-"):
        sign = -1.0
        s = s[1:]

    if not s or not _FULL_RE.fullmatch(s):
        raise ValueError(f"invalid duration string: {text!r}")

    total = 0.0
    last_order = -1
    seen = set()
    for number, unit in _TOKEN_RE.findall(s):
        order = _UNIT_ORDER[unit]
        if order <= last_order:
            raise ValueError(f"units out of order or repeated in {text!r}")
        if unit in seen:
            raise ValueError(f"repeated unit {unit!r} in {text!r}")
        seen.add(unit)
        last_order = order
        total += float(number) * _FACTORS[unit]

    return sign * total
```
