export class LruTtlCache {
  constructor({ capacity, defaultTtlMs = Infinity, now = () => Date.now() }) {
    if (!Number.isInteger(capacity) || capacity < 1) {
      throw new RangeError('capacity must be an integer >= 1');
    }
    this.capacity = capacity;
    this.defaultTtlMs = defaultTtlMs;
    this.now = now;
    this.map = new Map(); // key -> {value, expiry}
  }

  _isExpired(entry) {
    return entry.expiry !== Infinity && this.now() >= entry.expiry;
  }

  set(key, value, ttlMs) {
    const ttl = ttlMs !== undefined ? ttlMs : this.defaultTtlMs;
    const expiry = ttl === Infinity ? Infinity : this.now() + ttl;
    // Remove existing key to update order
    if (this.map.has(key)) {
      this.map.delete(key);
    }
    this.map.set(key, { value, expiry });

    // Evict if over capacity
    while (this.map.size > this.capacity) {
      const firstKey = this.map.keys().next().value;
      const firstEntry = this.map.get(firstKey);
      if (this._isExpired(firstEntry)) {
        this.map.delete(firstKey);
        continue;
      }
      // Evict least recently used
      this.map.delete(firstKey);
      break;
    }
    return this;
  }

  get(key) {
    const entry = this.map.get(key);
    if (!entry) return undefined;
    if (this._isExpired(entry)) {
      this.map.delete(key);
      return undefined;
    }
    // Move to most recently used
    this.map.delete(key);
    this.map.set(key, entry);
    return entry.value;
  }

  has(key) {

    if (!entry) return false;

      this.map.delete(key);
      return false;
    }

        this.map.delete(key);
      }
