```
import re

def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError("text must be a string")
    s = text.strip()
    if not s:
        raise ValueError("empty duration string")
    negative = False
    if s[0] == '-':
        negative = True
        s = s[1:]
    if not s:
        raise ValueError("duration string missing components")
    # pattern
    pattern = re.compile(r'(\d+(?:\.\d+)?)(ms|s|m|h)')
    unit_order = {'h':0, 'm':1, 's':2, 'ms':3}
    unit_factor = {'h':3600.0, 'm':60.0, 's':1.0, 'ms':0.001}
    seen = set()
    total = 0.0
    idx = 0
    last_order = -1
    while idx < len(s):
        m = pattern.match(s, idx)
        if not m:
            raise ValueError(f"invalid component at position {idx}")
        number_str, unit = m.group(1), m.group(2)
        # number conversion
        try:
            number = float(number_str)
        except ValueError:
            raise ValueError(f"invalid number '{number_str}'")
        if unit in seen:
            raise ValueError(f"repeated unit '{unit}'")
        seen.add(unit)
        order = unit_order[unit]
        if order <= last_order:
            raise ValueError(f"unit order violation: '{unit}' after previous unit")
        last_order = order
        total += number * unit_factor[unit]
        idx = m.end()
    if idx != len(s):
        raise ValueError("invalid trailing characters")
    if negative:
        total = -total
    return total
```
