def parse_duration(text):
    if not isinstance(text, str):
        raise TypeError("text must be a string")
    s = text.strip()
    if not s:
        raise ValueError("empty duration")
    # internal whitespace check
    if any(ch.isspace() for ch in s):
        raise ValueError("whitespace not allowed inside duration")
    sign = 1
    if s[0] == '-':
        sign = -1
        s = s[1:]
        if not s:
            raise ValueError("no components after sign")
    total = 0.0
    last_index = -1
    units = ['h','m','s','ms']
    factors = {'h':3600.0,'m':60.0,'s':1.0,'ms':0.001}
    pattern = re.compile(r'\d+(?:\.\d+)?')
    while s:
        m = pattern.match(s)
        if not m:
            raise ValueError(f"expected number at: {s}")
        num_str = m.group(0)
        try:
            value = float(num_str)
        except ValueError:
            raise ValueError("invalid number")
        s = s[m.end():]
        # unit detection
        if s.startswith('ms'):
            unit = 'ms'
            s = s[2:]
        elif s and s[0] in ('h','m','s'):
            unit = s[0]
            s = s[1:]
        else:
            raise ValueError(f"expected unit after number at: {s}")
        idx = units.index(unit)
        if idx <= last_index:
            raise ValueError("units out of order or repeated")
        total += value * factors[unit]
        last_index = idx
    return sign * total
