export class LruTtlCache {
  constructor({ capacity, defaultTtlMs = Infinity, now = () => Date.now() }) {
    if (!Number.isInteger(capacity) || capacity < 1) {
      throw new RangeError('capacity must be an integer ≥ 1');
    }
    this.capacity = capacity;
    this.defaultTtlMs = defaultTtlMs;
    this.now = now;
    this.map = new Map(); // key -> { value, expiryTime }
    this.accessOrder = new Map(); // key -> accessTime (for LRU)
  }

  #purgeExpired() {
    const now = this.now();
    for (const [key, entry] of this.map.entries()) {
      if (now >= entry.expiryTime) {
        this.map.delete(key);
        this.accessOrder.delete(key);
      }
    }
  }

  #evictIfNeeded() {
    // First, purge expired entries
    this.#purgeExpired();
    
    // If still over capacity, evict LRU entries
    while (this.map.size > this.capacity) {
      // Get the oldest access time
      let oldestTime = Infinity;
      let oldestKey = null;
      
      for (const [key, accessTime] of this.accessOrder.entries()) {
        if (accessTime < oldestTime) {
          oldestTime = accessTime;
          oldestKey = key;
        }
      }
      
      if (oldestKey !== null) {
        this.map.delete(oldestKey);
        this.accessOrder.delete(oldestKey);
      } else {
        break; // Safety check
      }
    }
  }

  set(key, value, ttlMs = this.defaultTtlMs) {
    const expiryTime = ttlMs === Infinity ? Infinity : this.now() + ttlMs;
    
    // Update access order
    const accessTime = this.now();
    this.accessOrder.set(key, accessTime);
    
    // Set or update the entry
    this.map.set(key, { value, expiryTime });
    
    // Evict if needed
    this.#evictIfNeeded();
    
    return this;
  }

  get(key) {
    const entry = this.map.get(key);
    if (entry === undefined) {
      return undefined;
    }
    
    const now = this.now();
    if (now >= entry.expiryTime) {
      this.map.delete(key);
      this.accessOrder.delete(key);
      return undefined;
    }
    
    // Update access order
    this.accessOrder.set(key, now);
    
    return entry.value;
  }

  has(key) {
    const entry = this.map.get(key);
    if (entry === undefined) {
      return false;
    }
    
    const now = this.now();
    if (now >= entry.expiryTime) {
      this.map.delete(key);
      this.accessOrder.delete(key);
      return false;
    }
    
    return true;
  }

  delete(key) {
    const deleted = this.map.delete(key);
    this.accessOrder.delete(key);
    return deleted;
  }

  get size() {
    this.#purgeExpired();
    return this.map.size;
  }

  keys() {
    this.#purgeExpired();
    // Create a copy of accessOrder sorted by access time (descending)
    const sortedEntries = Array.from(this.accessOrder.entries())
      .sort((a, b) => b[1] - a[1]);
    
    return sortedEntries.map(([key]) => key);
  }
}
