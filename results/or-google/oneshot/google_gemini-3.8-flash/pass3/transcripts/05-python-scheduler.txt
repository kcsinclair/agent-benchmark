"""Weighted Interval Scheduling at scale."""

import bisect


def best_schedule(
    jobs: list[tuple[float, float, float]],
) -> tuple[float, list[int]]:
    """Compute the maximum weight subset of non-overlapping jobs.

    Parameters
    ----------
    jobs : list[tuple[float, float, float]]
        A list of jobs where each job is represented as (start, end, weight).

    Returns
    -------
    tuple[float, list[int]]
        A tuple (total_weight, chosen_indices) where total_weight is the maximum
        achievable weight and chosen_indices are the indices of the selected jobs
        sorted by start time (breaking ties by index ascending).
    """
    if not jobs:
        return 0.0, []

    n = len(jobs)

    # Decorate with original index and sort primarily by end time
    sorted_jobs = sorted(
        ((start, end, weight, i) for i, (start, end, weight) in enumerate(jobs)),
        key=lambda x: x[1],
    )

    end_times = [job[1] for job in sorted_jobs]

    # dp[k] stores the maximum weight achievable using a subset of jobs from sorted_jobs[:k]
    dp = [0.0] * (n + 1)
    take = bytearray(n)
    p = [0] * n

    for i in range(n):
        start, _end, weight, _orig_idx = sorted_jobs[i]
        # Find the latest job that ends at or before this job starts
        p_i = bisect.bisect_right(end_times, start) - 1
        p[i] = p_i

        incl_weight = weight + dp[p_i + 1]
        if incl_weight > dp[i]:
            dp[i + 1] = incl_weight
            take[i] = 1
        else:
            dp[i + 1] = dp[i]

    # Iterative backtracking to reconstruct the chosen jobs
    curr = n
    chosen_indices = []
    while curr > 0:
        if take[curr - 1]:
            chosen_indices.append(sorted_jobs[curr - 1][3])
            curr = p[curr - 1] + 1
        else:
            curr -= 1

    # Sort chosen indices by original start time, tie-breaking by index ascending
    chosen_indices.sort(key=lambda idx: (jobs[idx][0], idx))

    return float(dp[n]), chosen_indices
