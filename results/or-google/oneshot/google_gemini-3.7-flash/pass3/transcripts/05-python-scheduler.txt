An optimal $O(n \log n)$ implementation of Weighted Interval Scheduling using dynamic programming and binary search (`bisect`).

```python
"""Weighted interval scheduling module."""

from bisect import bisect_right


def best_schedule(jobs: list[tuple[float, float, float]]) -> tuple[float, list[int]]:
    """Find a subset of non-overlapping jobs that maximizes total weight.

    Parameters
    ----------
    jobs : list[tuple[float, float, float]]
        A list of (start, end, weight) tuples.

    Returns
    -------
    tuple[float, list[int]]
        A tuple of (total_weight, chosen_indices), where chosen_indices are
        the original indices sorted by start time (ties broken by index ascending).
    """
    if not jobs:
        return 0.0, []

    n = len(jobs)

    # Decorate with original index and sort by end time
    # Tuple format: (start, end, weight, orig_idx)
    sorted_jobs = sorted(
        ((start, end, weight, i) for i, (start, end, weight) in enumerate(jobs)),
        key=lambda x: (x[1], x[0], x[3]),
    )

    end_times = [job[1] for job in sorted_jobs]

    # dp[i] represents the maximum weight achievable considering sorted_jobs[0..i-1]
    dp = [0.0] * (n + 1)
    take = [False] * (n + 1)
    prev_compat = [0] * (n + 1)

    for i in range(1, n + 1):
        start, _end, weight, _orig_idx = sorted_jobs[i - 1]

        # Find the latest job ending at or before `start`
        # bisect_right returns the count of jobs with end <= start
        p = bisect_right(end_times, start)
        prev_compat[i] = p

        incl_weight = dp[p] + weight
        if incl_weight > dp[i - 1]:
            dp[i] = incl_weight
            take[i] = True
        else:
            dp[i] = dp[i - 1]
            take[i] = False

    # Reconstruct optimal subset of jobs
    chosen_indices: list[int] = []
    curr = n
    while curr > 0:
        if take[curr]:
            chosen_indices.append(sorted_jobs[curr - 1][3])
            curr = prev_compat[curr]
        else:
            curr -= 1

    # Sort chosen indices by start time, breaking ties by index ascending
    chosen_indices.sort(key=lambda idx: (jobs[idx][0], idx))

    return float(dp[n]), chosen_indices
```
