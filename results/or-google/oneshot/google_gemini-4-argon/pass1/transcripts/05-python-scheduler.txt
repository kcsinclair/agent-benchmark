# scheduler.py
from bisect import bisect_right


def best_schedule(jobs: list[tuple[float, float, float]]) -> tuple[float, list[int]]:
    """
    Finds an optimal non-overlapping subset of weighted intervals (jobs).

    Args:
        jobs: A list of (start, end, weight) tuples.

    Returns:
        A tuple (total_weight, chosen_indices) where total_weight is the maximum
        achievable sum of weights as a float, and chosen_indices is a list of
        indices into the input list of the chosen jobs, sorted by start time.
    """
    n = len(jobs)
    if n == 0:
        return 0.0, []

    # Sort jobs by end time ascending.
    # Store as (end, start, weight, original_index) for fast tuple sorting.
    sorted_jobs = [(end, start, weight, idx) for idx, (start, end, weight) in enumerate(jobs)]
    sorted_jobs.sort(key=lambda x: x[0])

    ends = [job[0] for job in sorted_jobs]

    # dp[i] holds the max weight using a subset of the first i sorted jobs (1-indexed).
    # dp[0] = 0.0 is the base case for 0 jobs.
    dp = [0.0] * (n + 1)
    # prev[i] stores the 1-based index of the previous compatible job for sorted job i-1.
    prev = [0] * (n + 1)

    for i, (end, start, weight, _) in enumerate(sorted_jobs, 1):
        # Find the number of jobs with end_time <= start.
        # Since p is a count, it directly corresponds to the 1-based index in `dp`.
        p = bisect_right(ends, start, 0, i - 1)
        prev[i] = p

        val_with = dp[p] + weight
        val_without = dp[i - 1]

        if val_with > val_without:
            dp[i] = val_with
        else:
            dp[i] = val_without

    # Backtrack iteratively to reconstruct the chosen subset of jobs.
    chosen_indices = []
    curr = n
    while curr > 0:
        # If the current optimal value is strictly greater than the previous one,
        # then the job at `curr - 1` must have been included.
        if dp[curr] > dp[curr - 1]:
            chosen_indices.append(sorted_jobs[curr - 1][3])
            curr = prev[curr]
        else:
            curr -= 1

    # Sort chosen indices by job's start time ascending, breaking ties by index ascending.
    chosen_indices.sort(key=lambda idx: (jobs[idx][0], idx))

    return float(dp[n]), chosen_indices
