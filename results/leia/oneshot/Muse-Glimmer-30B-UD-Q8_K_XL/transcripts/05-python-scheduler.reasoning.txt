def best_schedule(jobs):
 if not jobs: return (0.0, [])
 n = len(jobs)
 # Build list
 items = []
 for idx, (s,e,w) in enumerate(jobs):
   items.append((e, s, w, idx))
 items.sort(key=lambda x: x[0]) # end
 ends = [it[0] for it in items]
 starts = [it[1] for it in items]
 weights = [it[2] for it in items]
 origs = [it[3] for it in items]

 # compute p using bisect_right
 import bisect
 p = [0]*n
 for i in range(n):
   s = starts[i]
   # find rightmost j < i with ends[j] <= s
   # bisect_right returns insertion point
   j = bisect.bisect_right(ends, s, 0, i) - 1
   p[i] = j  # -1 means none

 dp = [0.0]*(n+1)
 # also store choice?
 # For reconstruction we need to know if we took i.
 # We can store predecessor pointer.
 # Simpler: store take boolean.
 take = [False]*n
 for i in range(1, n+1):
   w_i = weights[i-1]
   j = p[i-1]
   incl = w_i + dp[j+1]  # because dp index offset
   excl = dp[i-1]
   if incl > excl + 1e-12: # tie? any optimal accepted. Might want deterministic?
     dp[i] = incl
     take[i-1] = True
   else:
     dp[i] = excl
     take[i-1] = False

 # reconstruct
 chosen = []
 i = n
 while i > 0:
   if take[i-1]:
     chosen.append(origs[i-1])
     i = p[i-1] + 1
   else:
     i -= 1
 chosen.reverse()
 # Now sort by start time then index
 # We have start times for chosen indices
 # Build list of (start, idx)
 # Could just sort
 # To avoid extra work, we can trust order is by end which implies start order.
 # But to guarantee tie break, sort.
 # Let's get start times from original jobs
 # Build list of tuples
 chosen_with_start = []
 for idx in chosen:
   s = jobs[idx][0]
   chosen_with_start.append((s, idx))
 chosen_with_start.sort(key=lambda x: (x[0], x[1]))
 chosen_sorted = [idx for _, idx in chosen_with_start]

 total = dp[n]
 # Ensure float
 return (float(total), chosen_sorted)
