   return []any{}, nil

 var mu sync.Mutex
 var wg sync.WaitGroup
 sem := make(chan struct{}, workers)

loop:

   if ctx.Err() != nil {
     break loop
   }
   mu.Lock()
   if firstErr != nil {
     mu.Unlock()
     break loop
   }
   mu.Unlock()

   case sem <- struct{}{}:
   case <-ctx.Done():
     break loop
   }

     mu.Unlock()
     <-sem
