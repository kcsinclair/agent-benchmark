We need to implement LRU cache with TTL. Use Map to store entries with order from most recent to least recent? In JS Map iteration order is insertion order. To implement LRU, we can move accessed/updated entries to the front (most recent). Since Map iteration order is insertion order, we can delete and re-set the key to move it to the end (most recent). But we need order from most recent to least recent for keys() output. If we treat end of Map as most recent, then keys() should return reversed order. Or we can treat beginning as most recent by using a doubly linked list. But requirement O(1) amortized and can use Map insertion order. Simpler: treat end as most recent. Then keys() should return array from most recent to least recent: iterate Map entries, collect keys, then reverse. That's O(n) for keys() which is fine.

We need to handle expiration: each entry stores value, expiryTime (number). When get/has, we check if expired (now >= expiry). If expired, delete entry and treat as not present.

size getter must purge expired entries before counting. Could compute by iterating and deleting expired entries, then return map.size.

When set, we compute ttl = ttlMs !== undefined ? ttlMs : defaultTtlMs. expiry = now() + ttl. If ttl is Infinity, expiry = Infinity. If ttl <=0, expiry = now() (or now() + ttl) which will be <= now, so considered expired; we can just not store? But spec says entry with ttl 0 or negative means already expired. So set should store? It would be considered expired and removed when discovered. Could just store and let later removal happen. But better to not store to keep capacity. We'll store but treat as expired; but then size may count it incorrectly until purge. Simpler: if ttl <=0, we can just delete any existing key and return this (no entry). That matches semantics.

When capacity exceeded: first evict expired entries. So we need a method to purge expired entries (maybe while loop). Then if still over capacity, evict least recently used (the oldest entry in Map). Since we treat end as most recent, the oldest is first entry (Map.prototype.keys().next().value). So we can delete that.

Implementation details:

class LruTtlCache {
  constructor({capacity, defaultTtlMs = Infinity, now = () => Date.now()}) {
    if (!Number.isInteger(capacity) || capacity < 1) throw new RangeError('capacity must be integer >=1');
    this._capacity = capacity;
    this._defaultTtlMs = defaultTtlMs;
    this._now = now;
    this._map = new Map(); // key -> {value, expiry}
  }

  _isExpired(entry) {
    const now = this._now();
    return entry.expiry !== Infinity && now >= entry.expiry;
  }

  _purgeExpired() {
    const now = this._now();
    for (const [key, entry] of this._map) {
      if (entry.expiry !== Infinity && now >= entry.expiry) {
        this._map.delete(key);
      }
    }
  }

  set(key, value, ttlMs) {
    const ttl = ttlMs !== undefined ? ttlMs : this._defaultTtlMs;
    const now = this._now();
    const expiry = ttl === Infinity ? Infinity : now + ttl;
    // If ttl <=0, treat as expired: delete any existing and do nothing
    if (ttl <= 0) {
      this._map.delete(key);
      return this;
    }
    // If key exists, delete to re-insert for recency
    if (this._map.has(key)) {
      this._map.delete(key);
    }
    this._map.set(key, {value, expiry});
    // Ensure capacity
    this._ensureCapacity();
    return this;
  }

  _ensureCapacity() {
    // First purge expired entries
    this._purgeExpired();
    while (this._map.size > this._capacity) {
      // evict least recently used: first entry
      const lruKey = this._map.keys().next().value;
      this._map.delete(lruKey);
    }
  }

  get(key) {
    const entry = this._map.get(key);
    if (!entry) return undefined;
    if (this._isExpired(entry)) {
      this._map.delete(key);
      return undefined;
    }
    // move to most recent
    this._map.delete(key);
    this._map.set(key, entry);
    return entry.value;
  }

  has(key) {
    const entry = this._map.get(key);
    if (!entry) return false;
    if (this._isExpired(entry)) {
      this._map.delete(key);
      return false;
    }
    return true;
  }

  delete(key) {
    return this._map.delete(key);
  }

  get size() {
    this._purgeExpired();
    return this._map.size;
  }

  keys() {
    // Return array from most recent to least recent
    const keys = Array.from(this._map.keys());
    // purge expired first? Should not include expired keys.
    // We'll filter while building.
    const now = this._now();
    const result = [];
    for (let i = keys.length - 1; i >= 0; i--) {
      const k = keys[i];
      const entry = this._map.get(k);
      if (entry && (entry.expiry === Infinity || now < entry.expiry)) {
        result.push(k);
      } else {
        // expired, remove
        this._map.delete(k);
      }
    }
    return result;
  }
}
