func Run(parent context.Context, tasks []Task, workers int) ([]any, error) {
    if workers < 1 {
        return nil, errors.New("workers must be >= 1")
    }
    if len(tasks) == 0 {
        return nil, nil
    }

    ctx, cancel := context.WithCancel(parent)
    defer cancel()

    type job struct {
        idx int
        task Task
    }

    taskCh := make(chan job)
    var wg sync.WaitGroup

    results := make([]any, len(tasks))

    var once sync.Once
    var firstErr error

    // worker function
    worker := func() {
        defer wg.Done()
        for {
            select {
            case <-ctx.Done():
                return
            case j, ok := <-taskCh:
                if !ok {
                    return
                }
                // run task with ctx (maybe we want to pass ctx)
                res, err := j.task(ctx)
                if err != nil {
                    once.Do(func() {
                        firstErr = err
                        cancel()
                    })
                    // don't store result
                    continue
                }
                results[j.idx] = res
            }
        }
    }

    // start workers
    if workers > len(tasks) {
        workers = len(tasks)
    }
    wg.Add(workers)
    for i := 0; i < workers; i++ {
        go worker()
    }

    // feed tasks
    go func() {
        defer close(taskCh)
        for i, t := range tasks {
            select {
            case <-ctx.Done():
                return
            case taskCh <- job{idx: i, task: t}:
            }
        }
    }()

    // wait for workers
    wg.Wait()

    // after workers done, check context error
    if firstErr != nil {
        return nil, firstErr
    }
    if err := ctx.Err(); err != nil && err != context.Canceled {
        // if parent cancelled, ctx.Err() may be context.Canceled or DeadlineExceeded
        return nil, err
    }
    return results, nil
}
