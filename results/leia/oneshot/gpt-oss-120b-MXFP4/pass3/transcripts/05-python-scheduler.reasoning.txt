def best_schedule(jobs):
    if not jobs:
        return (0.0, [])
    enriched = [(s, e, w, i) for i, (s, e, w) in enumerate(jobs)]
    enriched.sort(key=lambda x: (x[1], x[0]))  # sort by end then start
    starts = [s for s, e, w, i in enriched]
    ends = [e for s, e, w, i in enriched]
    weights = [w for s, e, w, i in enriched]
    orig = [i for s, e, w, i in enriched]

    import bisect
    p = [0]*len(jobs)
    for idx in range(len(jobs)):
        # find rightmost end <= start[idx]
        j = bisect.bisect_right(ends, starts[idx]) - 1
        p[idx] = j
    n = len(jobs)
    dp = [0.0]*(n+1)
    take = [False]*n
    for i in range(1, n+1):
        w = weights[i-1]
        incl = w + dp[p[i-1]+1]  # if p=-1 then dp[0]
        excl = dp[i-1]
        if incl > excl + 1e-12:
            dp[i] = incl
            take[i-1] = True
        else:
            dp[i] = excl
            take[i-1] = False
    # reconstruct
    chosen = []
    i = n
    while i > 0:
        if take[i-1]:
            chosen.append(orig[i-1])
            i = p[i-1] + 1
        else:
            i -= 1
    # sort chosen by start then index
    chosen.sort(key=lambda idx: (jobs[idx][0], idx))
    total = dp[n]
    return (total, chosen)
