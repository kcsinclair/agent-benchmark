func Run(ctx context.Context, tasks []Task, workers int) ([]any, error) {
    if workers < 1 {
        return nil, errors.New("workers must be >= 1")
    }
    if len(tasks) == 0 {
        return nil, nil
    }

    // Create a child context that we can cancel on first error or parent cancel.
    taskCtx, cancelAll := context.WithCancel(ctx)
    defer cancelAll() // ensure resources freed

    type job struct {
        idx int
        fn Task
    }

    jobs := make(chan job)
    // results slice
    results := make([]any, len(tasks))

    var wg sync.WaitGroup
    // error handling
    var once sync.Once
    var firstErr error

    // worker function
    worker := func() {
        defer wg.Done()
        for j := range jobs {
            // before executing, check if context already done
            select {
            case <-taskCtx.Done():
                // don't start new tasks
                // we could skip execution
                // but we need to ensure we don't block; just continue to next iteration (which will break when channel closed)
                continue
            default:
            }

            res, err := j.fn(taskCtx)
            if err != nil {
                once.Do(func() {
                    firstErr = err
                    // cancel all tasks
                    cancelAll()
                })
                // we still need to store maybe nil? Not needed because we will return nil results on error.
                // continue to drain other jobs (but they may be prevented from starting)
                continue
            }
            // store result if no error yet
            // but if error already occurred, we can ignore storing.
            // However ordering requirement only on success.
            // So we can store anyway; not used.
            results[j.idx] = res
        }
    }

    // start workers
    wg.Add(workers)
    for i := 0; i < workers; i++ {
        go worker()
    }

    // feed jobs
    go func() {
        defer close(jobs)
        for i, t := range tasks {
            // before sending, check if context done (error or parent cancel)
            select {
            case <-taskCtx.Done():
                // stop sending further jobs
                return
            default:
                // send
                jobs <- job{idx: i, fn: t}
            }
        }
    }()

    // Wait for workers to finish
    wg.Wait()

    // After workers done, check errors
    if firstErr != nil {
        return nil, firstErr
    }
    // also check if ctx cancelled
    if err := ctx.Err(); err != nil {
        return nil, err
    }
    // success
    return results, nil
}
