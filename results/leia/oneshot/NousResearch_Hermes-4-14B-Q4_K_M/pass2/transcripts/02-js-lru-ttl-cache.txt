import { Map } from 'node:map';

export class LruTtlCache {
  constructor({ capacity, defaultTtlMs = Infinity, now = () => Date.now() }) {
    if (typeof capacity !== 'number' || capacity < 1 || !Number.isInteger(capacity)) {
      throw new RangeError('Capacity must be an integer >= 1');
    }
    this.capacity = capacity;
    this.defaultTtlMs = defaultTtlMs;
    this.now = now;
    this.map = new Map();
    this.expiryMap = new Map();
  }

  set(key, value, ttlMs) {
    const ttl = ttlMs ?? this.defaultTtlMs;
    const expiryTime = ttl === Infinity ? Infinity : this.now() + ttl;

    if (this.map.has(key)) {
      this.map.delete(key);
      this.expiryMap.delete(key);
    }

    this.map.set(key, value);
    this.expiryMap.set(key, expiryTime);

    this._purgeExpired();
    this._evictIfOverCapacity();

    return this;
  }

  get(key) {
    const value = this.map.get(key);
    if (value === undefined) return undefined;

    const expiryTime = this.expiryMap.get(key);
    if (expiryTime === Infinity) return value;
    if (this.now() >= expiryTime) {
      this.map.delete(key);
      this.expiryMap.delete(key);
      return undefined;
    }

    this.map.delete(key);
    this.map.set(key, value);
    return value;
  }

  has(key) {
    const expiryTime = this.expiryMap.get(key);
    if (expiryTime === undefined) return false;

    if (expiryTime === Infinity) return true;
    if (this.now() >= expiryTime) {
      this.map.delete(key);
      this.expiryMap.delete(key);
      return false;
    }
    return true;
  }

  delete(key) {
    if (this.map.has(key)) {
      this.map.delete(key);
      this.expiryMap.delete(key);
      return true;
    }
    return false;
  }

  get size() {
    this._purgeExpired();
    return this.map.size;
  }

  keys() {
    this._purgeExpired();
    return Array.from(this.map.keys());
  }

  _purgeExpired() {
    const now = this.now();
    for (const [key, expiryTime] of this.expiryMap) {
      if (expiryTime === Infinity) continue;
      if (now >= expiryTime) {
        this.map.delete(key);
        this.expiryMap.delete(key);
      }
    }
  }

  _evictIfOverCapacity() {
    while (this.map.size > this.capacity) {
      const [key] = this.map.keys();
      this.map.delete(key);
      this.expiryMap.delete(key);
    }
  }
}
