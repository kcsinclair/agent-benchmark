import bisect

def best_schedule(jobs: list[tuple[float, float, float]]) -> tuple[float, list[int]]:
    if not jobs:
        return (0.0, [])

    # Sort jobs by end time
    jobs_sorted = sorted(jobs, key=lambda x: x[1])
    n = len(jobs_sorted)
    end_times = [job[1] for job in jobs_sorted]

    # Precompute p[j] for each job j (0-based index)
    p = [0] * n
    for j in range(n):
        # Find the latest job that ends before or at jobs_sorted[j][0]
        start_j = jobs_sorted[j][0]
        # Use bisect_right to find insertion point of start_j in end_times
        # This gives the index of the first end time > start_j
        # So the latest job is at index idx-1
        idx = bisect.bisect_right(end_times, start_j)
        p[j] = idx - 1  # -1 if no such job

    # Dynamic programming arrays
    # dp[j] = max weight for jobs 0..j
    # prev[j] = predecessor job index for optimal solution up to j
    dp = [0.0] * n
    prev = [-1] * n

    dp[0] = jobs_sorted[0][2]  # weight of first job

    for j in range(1, n):
        # Option 1: include jobs_sorted[j]
        weight_j = jobs_sorted[j][2]
        # Find the latest compatible job using p[j]
        if p[j] >= 0:
            weight_with_j = dp[p[j]] + weight_j
        else:
            weight_with_j = weight_j

        # Option 2: exclude jobs_sorted[j]
        weight_without_j = dp[j-1]

        if weight_with_j > weight_without_j:
            dp[j] = weight_with_j
            prev[j] = p[j]  # predecessor is p[j]
        else:
            dp[j] = weight_without_j
            prev[j] = j-1   # predecessor is j-1

    max_weight = dp[n-1]
    # Reconstruct the chosen indices
    chosen_indices = []
    current = n-1
    while current != -1:
        chosen_indices.append(current)
        current = prev[current]

    # Sort chosen_indices by start time (and then by index if needed)
    # Since jobs_sorted is sorted by end time, and we reconstruct from dp,
    # the chosen_indices are already in order of increasing end time.
    # To sort by start time, we need to map back to original indices.
    # However, the problem requires the indices to be sorted by start time.
    # We need to sort the chosen_indices based on the start time of the jobs.
    # Since jobs_sorted is sorted by end time, the chosen_indices are not necessarily sorted by start time.
    # We need to sort them by the start time of the jobs in the original list.
    # But the problem states: "chosen_indices are the indices into the input list of one optimal subset, sorted by the job's start time (break ties by index ascending)."
    # So we need to sort the chosen_indices based on the start time of the jobs in the original list.
    # However, jobs_sorted is a sorted version of the input list. The original indices are lost.
    # We need to store the original indices when sorting jobs_sorted.
    # Let's re-sort the jobs with their original indices.
    # This is a critical point. The current approach reconstructs indices in the sorted list, not the original list.
    # We need to track original indices throughout.

    # Revised plan:
    # 1. Sort jobs by end time, but keep track of original indices.
    # 2. When reconstructing, we have the original indices.
    # 3. Then, sort these original indices by their start time (and then by index).

    # Let's redo the sorting to include original indices.
    # jobs_sorted_with_indices = sorted(enumerate(jobs), key=lambda x: x[1][1])
    # This gives a list of (original_index, (start, end, weight))
    # Then, when we process, we use these.

    # Let's refactor the code to include original indices.

    # Step 1: Sort jobs by end time, keeping original indices
    jobs_with_indices = list(enumerate(jobs))
    jobs_with_indices.sort(key=lambda x: x[1][1])  # sort by end time
    n = len(jobs_with_indices)
    end_times = [job[1][1] for job in jobs_with_indices]
    original_indices = [job[0] for job in jobs_with_indices]

    # Precompute p[j] for each job j (0-based index in jobs_with_indices)
    p = [0] * n
    for j in range(n):
        start_j = jobs_with_indices[j][1][0]
        idx = bisect.bisect_right(end_times, start_j)
        p[j] = idx - 1  # -1 if no such job

    # Dynamic programming arrays
    dp = [0.0] * n
    prev = [-1] * n

    dp[0] = jobs_with_indices[0][1][2]  # weight of first job

    for j in range(1, n):
        weight_j = jobs_with_indices[j][1][2]
        if p[j] >= 0:
            weight_with_j = dp[p[j]] + weight_j
        else:
            weight_with_j = weight_j

        weight_without_j = dp[j-1]

        if weight_with_j > weight_without_j:
            dp[j] = weight_with_j
            prev[j] = p[j]
        else:
            dp[j] = weight_without_j
            prev[j] = j-1

    max_weight = dp[n-1]
    # Reconstruct the chosen indices in jobs_with_indices
    chosen_indices_in_sorted = []
    current = n-1
    while current != -1:
        chosen_indices_in_sorted.append(current)
        current = prev[current]

    # Now, map these indices back to original indices
    chosen_original_indices = [original_indices[i] for i in chosen_indices_in_sorted]

    # Sort chosen_original_indices by start time (and then by index)
    # We need to sort by the start time of the jobs in the original list.
    # We can do this by sorting the list of original indices based on the start time of the corresponding job.
    # We have the original list of jobs. We can create a list of tuples (start, index) for the chosen jobs.
    # Then sort by start, then by index.
    chosen_original_indices.sort(key=lambda idx: (jobs[idx][0], idx))

    return (max_weight, chosen_original_indices)
