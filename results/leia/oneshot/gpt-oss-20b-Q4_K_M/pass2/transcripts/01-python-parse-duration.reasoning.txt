import re

def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError("text must be a string")
    s = text.strip()
    if not s:
        raise ValueError("empty duration")
    # check for internal whitespace
    if any(c.isspace() for c in s):
        raise ValueError("whitespace inside duration")
    negative = False
    if s[0] == '-':
        negative = True
        s = s[1:]
        if not s:
            raise ValueError("no duration after minus")
    # regex pattern
    pattern = re.compile(r'(\d+(?:\.\d+)?)(h|m|s|ms)')
    pos = 0
    total = 0.0
    last_index = -1
    unit_order = {'h':0,'m':1,'s':2,'ms':3}
    multiplier = {'h':3600.0,'m':60.0,'s':1.0,'ms':0.001}
    while pos < len(s):
        m = pattern.match(s, pos)
        if not m:
            raise ValueError(f"invalid component at position {pos}")
        number_str, unit = m.group(1), m.group(2)
        # check order
        idx = unit_order[unit]
        if idx <= last_index:
            raise ValueError(f"unit order error: {unit}")
        last_index = idx
        # convert number
        try:
            number = float(number_str)
        except ValueError:
            raise ValueError(f"invalid number: {number_str}")
        total += number * multiplier[unit]
        pos = m.end()
    if pos != len(s):
        raise ValueError("unparsed trailing characters")
    if negative:
        total = -total
    return total
