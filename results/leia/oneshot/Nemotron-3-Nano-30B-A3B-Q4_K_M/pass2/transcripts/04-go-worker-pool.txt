```go
package main

import (
	"context"
	"errors"
	"sync"
)

// Run executes tasks with at most `workers` running concurrently and returns
// their results in the same order as the input slice.
func Run(ctx context.Context, tasks []Task, workers int) ([]any, error) {
	if workers < 1 {
		return nil, errors.New("workers must be >= 1")
	}
	if len(tasks) == 0 {
		return []any{}, nil
	}

	// result holds the output values in order.
	results := make([]any, len(tasks))
	// errMu protects the first error that is returned.
	var errMu struct {
		sync.Mutex
		err error
	}
	// errMu.err will store the first non-nil error encountered.

	// sem limits the number of concurrently running tasks.
	sem := make(chan struct{}, workers)

	// wg waits for all started tasks to finish.
	var wg sync.WaitGroup

	// cancel stops new tasks from starting when the context ends.
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()

	// start each task.
	for i, task := range tasks {
		// If the context is already done, abort immediately.
		if ctx.Err() != nil {
			break
		}
		// Acquire a semaphore slot.
		select {
		case <-ctx.Done():
			break
		case sem <- struct{}{}:
		}
		// If we couldn't acquire a slot because of context cancellation,
		// release any already-acquired slots and exit.
		if ctx.Err() != nil {
			// Drain the semaphore to avoid leaks.
			for i := 0; i < workers; i++ {
				select {
				case <-sem:
				default:
				}
			}
			break
		}

		wg.Add(1)
		// Run the task in a goroutine.
		go func(i int, task Task) {
			defer wg.Done()
			// Each task gets its own derived context that can be cancelled
			// when the first error occurs or when the parent context ends.
			taskCtx, taskCancel := context.WithCancel(ctx)
			defer taskCancel()

			// If an error has already been recorded, skip execution.
			errMu.Lock()
			if errMu.err != nil {
				errMu.Unlock()
				// Ensure we still release the semaphore slot.
				select {
				case sem <- struct{}{}:
				default:
				}
				return
			}
			errMu.Unlock()

			// Execute the task.
			res, err := task(taskCtx)

			// If the task returned an error, record it (first one wins).
			if err != nil {
				errMu.Lock()
				if errMu.err == nil {
					errMu.err = err
				}
				errMu.Unlock()

				// Cancel the derived context to abort other tasks.
				taskCancel()
				// Ensure the semaphore slot is released.
				select {
				case sem <- struct{}{}:
				default:
				}
				return
			}

			// Store the result in the correct position.
			results[i] = res

			// If an error was recorded by another task, cancel this task.
			errMu.Lock()
			if errMu.err != nil {
				taskCancel()
			}
			errMu.Unlock()
		}(i, task)
	}

	// Wait for all started tasks to finish.
	wg.Wait()

	// If the original context was cancelled or timed out, return that error.
	if ctx.Err() != nil {
		return nil, ctx.Err()
	}

	// If any task failed, return the first error.
	errMu.Lock()
	err := errMu.err
	errMu.Unlock()
	if err != nil {
		return nil, err
	}
	return results, nil
}
```
