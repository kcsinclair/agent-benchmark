### q1.sql
```sql
SELECT 
    c.name AS customer_name,
    ROUND(SUM(oi.quantity * p.unit_price), 2) AS revenue
FROM customers c
JOIN orders o ON c.id = o.customer_id
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE o.status = 'completed'
GROUP BY c.id, c.name
ORDER BY revenue DESC, customer_name ASC;
```

### q2.sql
```sql
SELECT 
    p.name AS product_name,
    SUM(oi.quantity) AS units_sold
FROM products p
JOIN order_items oi ON p.id = oi.product_id
JOIN orders o ON oi.order_id = o.id
WHERE o.status = 'completed'
GROUP BY p.id, p.name
ORDER BY units_sold DESC, product_name ASC
LIMIT 3;
```

### q3.sql
```sql
SELECT 
    name AS customer_name,
    country
FROM customers
WHERE id NOT IN (SELECT DISTINCT customer_id FROM orders)
ORDER BY customer_name ASC;
```

### q4.sql
```sql
SELECT 
    strftime('%Y-%m', o.ordered_at) AS month,
    ROUND(SUM(oi.quantity * p.unit_price), 2) AS revenue
FROM orders o
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE o.status = 'completed' AND o.ordered_at LIKE '2025%'
GROUP BY month
ORDER BY month ASC;
```
