```python
import re


def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError(f"Expected str, got {type(text).__name__}")
    
    text = text.strip()
    
    if not text:
        raise ValueError("Empty duration string")
    
    # Check for leading minus
    negative = False
    if text.startswith('-'):
        negative = True
        text = text[1:]
    
    if not text:
        raise ValueError("No components after minus sign")
    
    # Check for any whitespace in the remaining text
    if any(c.isspace() for c in text):
        raise ValueError("Whitespace not allowed in duration string")
    
    # Define the pattern for a single component: number followed by unit
    # Number: digits with optional fractional part (no leading +, no exponents, no bare ., no trailing dot)
    # Unit: h, m, s, ms
    # We need to match components in order: h, m, s, ms
    
    # Let's use a regex to find all components
    # Each component is: (\d+(?:\.\d+)?)(h|ms|m|s)
    # But we need to be careful about the order and that each unit appears at most once
    
    # Let's find all matches
    pattern = r'(\d+(?:\.\d+)?)(h|ms|m|s)'
    matches = list(re.finditer(pattern, text))
    
    if not matches:
        raise ValueError("No valid components found")
    
    # Check that the entire string is consumed by the matches
    # The matches should cover the entire text
    pos = 0
    for match in matches:
        if match.start() != pos:
            raise ValueError(f"Invalid character at position {pos}")
        pos = match.end()
    
    if pos != len(text):
        raise ValueError(f"Invalid character at position {pos}")
    
    # Now validate the order and uniqueness
    unit_order = {'h': 0, 'm': 1, 's': 2, 'ms': 3}
    seen_units = set()
    last_order = -1
    
    total_seconds = 0.0
    
    for match in matches:
        number_str = match.group(1)
        unit = match.group(2)
        
        if unit in seen_units:
            raise ValueError(f"Repeated unit: {unit}")
        
        order = unit_order[unit]
        if order <= last_order:
            raise ValueError(f"Units not in descending order: {unit} after previous")
        
        seen_units.add(unit)
        last_order = order
        
        number = float(number_str)
        
        if unit == 'h':
            total_seconds += number * 3600
        elif unit == 'm':
            total_seconds += number * 60
        elif unit == 's':
            total_seconds += number
        elif unit == 'ms':
            total_seconds += number / 1000
    
    if negative:
        total_seconds = -total_seconds
    
    return total_seconds
```
