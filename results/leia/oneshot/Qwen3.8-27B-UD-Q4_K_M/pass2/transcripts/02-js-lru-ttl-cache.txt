export class LruTtlCache {
  constructor({ capacity, defaultTtlMs = Infinity, now = () => Date.now() }) {
    if (!Number.isInteger(capacity) || capacity < 1) {
      throw new RangeError('capacity must be an integer >= 1');
    }
    this._capacity = capacity;
    this._defaultTtlMs = defaultTtlMs;
    this._now = now;
    // Map preserves insertion order; we use it for both storage and LRU ordering.
    // Most recently used is at the end, least recently used is at the beginning.
    this._map = new Map();
  }

  _isExpired(entry) {
    if (entry.expiry === Infinity) return false;
    return this._now() >= entry.expiry;
  }

  _purgeExpired() {
    // Remove all expired entries from the map.
    // We iterate and collect keys to delete to avoid modifying during iteration issues.
    const toDelete = [];
    for (const [key, entry] of this._map) {
      if (this._isExpired(entry)) {
        toDelete.push(key);
      }
    }
    for (const key of toDelete) {
      this._map.delete(key);
    }
  }

  _evictIfNeeded() {
    // If size exceeds capacity, evict least recently used (first entry in map)
    // until we're within capacity.
    while (this._map.size > this._capacity) {
      // The first key in the map is the least recently used
      const firstKey = this._map.keys().next().value;
      this._map.delete(firstKey);
    }
  }

  set(key, value, ttlMs) {
    const ttl = ttlMs !== undefined ? ttlMs : this._defaultTtlMs;
    const expiry = ttl === Infinity ? Infinity : this._now() + ttl;

    // If key already exists, remove it first so we can re-insert at the end (most recent)
    if (this._map.has(key)) {
      this._map.delete(key);
    }

    // Insert at the end (most recently used)
    this._map.set(key, { value, expiry });

    // Purge expired entries first, then evict LRU if still over capacity
    this._purgeExpired();
    this._evictIfNeeded();

    return this;
  }

  get(key) {
    const entry = this._map.get(key);
    if (entry === undefined) {
      return undefined;
    }

    if (this._isExpired(entry)) {
      this._map.delete(key);
      return undefined;
    }

    // Move to most recently used: delete and re-insert
    this._map.delete(key);
    this._map.set(key, entry);

    return entry.value;
  }

  has(key) {
    const entry = this._map.get(key);
    if (entry === undefined) {
      return false;
    }

    if (this._isExpired(entry)) {
      this._map.delete(key);
      return false;
    }

    return true;
  }

  delete(key) {
    const had = this._map.delete(key);
    return had;
  }

  get size() {
    // Purge expired entries and return the count of live entries
    this._purgeExpired();
    return this._map.size;
  }

  keys() {
    // Return live keys ordered from most recently used to least recently used
    // Map iteration is from least recently used (first) to most recently used (last)
    // So we need to reverse the order
    this._purgeExpired();
    const keys = Array.from(this._map.keys());
    keys.reverse();
    return keys;
  }
}
