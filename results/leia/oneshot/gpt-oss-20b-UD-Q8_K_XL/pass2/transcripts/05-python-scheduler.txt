#!/usr/bin/env python3
"""
Weighted interval scheduling – O(n log n) implementation.

The function `best_schedule` receives a list of jobs, each represented as
(start, end, weight).  It returns the maximum total weight that can be
achieved by selecting a subset of non‑overlapping jobs, together with one
optimal subset of job indices sorted by start time (ties broken by index).

The algorithm sorts the jobs by end time, pre‑computes for each job the
index of the last compatible job using binary search, and then runs a
classic dynamic‑programming recurrence in linear time.  Reconstruction
is performed by following the decisions stored during the DP phase.
"""

from __future__ import annotations

from bisect import bisect_right
from typing import List, Tuple


def best_schedule(jobs: List[Tuple[float, float, float]]) -> Tuple[float, List[int]]:
    """
    Return the maximum total weight and one optimal subset of job indices.

    Parameters
    ----------
    jobs : list[tuple[float, float, float]]
        Each tuple is (start, end, weight).  All starts < ends and all
        weights > 0.

    Returns
    -------
    tuple[float, list[int]]
        (total_weight, chosen_indices)

        * total_weight : float – the maximum achievable sum of weights.
        * chosen_indices : list[int] – indices into the original `jobs`
          list of one optimal subset, sorted by job start time
          (ties broken by index ascending).

    Notes
    -----
    * If `jobs` is empty, returns `(0.0, [])`.
    * The algorithm runs in O(n log n) time and O(n) memory.
    """
    n = len(jobs)
    if n == 0:
        return 0.0, []

    # Attach original indices and sort by end time
    jobs_with_idx = [(s, e, w, idx) for idx, (s, e, w) in enumerate(jobs)]
    jobs_with_idx.sort(key=lambda x: x[1])  # sort by end

    # Unpack sorted data
    starts = [s for s, _, _, _ in jobs_with_idx]
    ends = [e for _, e, _, _ in jobs_with_idx]
    weights = [w for _, _, w, _ in jobs_with_idx]
    orig_idx = [idx for _, _, _, idx in jobs_with_idx]

    # Pre‑compute p[i] – index of last job that ends <= start of job i
    # (using 0‑based indices for jobs_with_idx)
    p = [0] * n
    for i in range(n):
        # bisect_right returns insertion point; subtract 1 to get last <= start
        j = bisect_right(ends, starts[i]) - 1
        p[i] = j  # may be -1 if no compatible job

    # DP arrays
    dp = [0.0] * (n + 1)          # dp[0] = 0
    take = [False] * (n + 1)      # take[i] indicates if job i-1 is taken

    for i in range(1, n + 1):
        w = weights[i - 1]
        j = p[i - 1] + 1          # dp index for compatible job
        incl = w + dp[j]
        if incl > dp[i - 1]:
            dp[i] = incl
            take[i] = True
        else:
            dp[i] = dp[i - 1]
            take[i] = False

    total_weight = dp[n]

    # Reconstruct chosen indices
    chosen = []
    i = n
    while i > 0:
        if take[i]:
            chosen.append(orig_idx[i - 1])
            i = p[i - 1] + 1
        else:
            i -= 1

    # Sort chosen indices by start time (ties by index)
    chosen.sort(key=lambda idx: (jobs[idx][0], idx))

    return total_weight, chosen
