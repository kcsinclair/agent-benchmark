    if stripped[0] == '-':

    # After removing sign, check for whitespace inside
    if re.search(r'\s', stripped):

    component_re = re.compile(r'(\d+(?:\.\d+)?)(h|m|s|ms)')

    seen_units = set()

    unit_order = {'h':0, 'm':1, 's':2, 'ms':3}

        idx = unit_order[unit]
        if idx <= last_index:
            raise ValueError("Units out of order or repeated")
        if unit in seen_units:
            raise ValueError("Repeated unit")
        seen_units.add(unit)
        last_index = idx
        # Convert number to float
        try:
            number = float(number_str)
        except ValueError:
            raise ValueError("Invalid number")
        # Multiply by factor
        if unit == 'h':
            factor = 3600.0
        elif unit == 'm':
            factor = 60.0
        elif unit == 's':
            factor = 1.0
        else:  # ms

        total_seconds += number * factor
        pos = m.end()
    # Ensure entire string consumed
    if pos != len(stripped):

    return sign * total_seconds
