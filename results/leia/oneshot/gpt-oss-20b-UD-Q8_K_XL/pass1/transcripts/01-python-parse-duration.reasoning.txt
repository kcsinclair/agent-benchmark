def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError("Input must be a string")
    s = text.strip()
    if not s:
        raise ValueError("Empty duration string")
    # regex to capture optional minus and components
    pattern = re.compile(r'^(-)?((?:\d+(?:\.\d+)?(?:h|m|s|ms))+)$')
    m = pattern.match(s)
    if not m:
        raise ValueError("Invalid format")
    sign = -1 if m.group(1) else 1
    components_str = m.group(2)
    # Now parse components sequentially
    # Use regex findall to get number and unit
    comp_pattern = re.compile(r'(\d+(?:\.\d+)?)(h|m|s|ms)')
    comps = comp_pattern.findall(components_str)
    if not comps:
        raise ValueError("No components found")
    # Ensure that concatenated components string equals original components_str
    # Because findall may skip overlapping? But pattern ensures full match. We can join.
    joined = ''.join(num+unit for num, unit in comps)
    if joined != components_str:
        raise ValueError("Invalid component sequence")
    # Now check order and uniqueness
    order_map = {'h':3,'m':2,'s':1,'ms':0}
    last_order = 4  # greater than any
    total_seconds = 0.0
    for num_str, unit in comps:
        order = order_map[unit]
        if order > last_order:
            raise ValueError("Units out of order")
        if order == last_order:
            raise ValueError("Repeated unit")
        last_order = order
        value = float(num_str)
        if unit == 'h':
            total_seconds += value * 3600
        elif unit == 'm':
            total_seconds += value * 60
        elif unit == 's':
            total_seconds += value
        elif unit == 'ms':
            total_seconds += value / 1000
    return sign * total_seconds
