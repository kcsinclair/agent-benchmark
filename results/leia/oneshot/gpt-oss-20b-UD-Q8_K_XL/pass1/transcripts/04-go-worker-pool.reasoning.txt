import (
    "context"
    "fmt"
    "sync"
)

type Task func(ctx context.Context) (any, error)

func Run(ctx context.Context, tasks []Task, workers int) ([]any, error) {
    if workers < 1 {
        return nil, fmt.Errorf("workers must be >= 1")
    }
    if len(tasks) == 0 {
        return []any{}, nil
    }

    ctx, cancel := context.WithCancel(ctx)
    defer cancel()

    results := make([]any, len(tasks))
    tokens := make(chan struct{}, workers)
    errorCh := make(chan error, 1)
    var wg sync.WaitGroup

    var firstErr error

    for idx, t := range tasks {
        // Check for cancellation or previous error before acquiring a token
        select {
        case <-ctx.Done():
            firstErr = ctx.Err()
            cancel()
            goto done
        case err := <-errorCh:
            firstErr = err
            cancel()
            goto done
        default:
        }

        // Acquire a token to respect the worker limit
        select {
        case <-ctx.Done():
            firstErr = ctx.Err()
            cancel()
            goto done
        case err := <-errorCh:
            firstErr = err
            cancel()
            goto done
        case tokens <- struct{}{}:
            // token acquired
        }

        wg.Add(1)
        go func(i int, task Task) {
            defer wg.Done()
            defer func() { <-tokens }()
            res, err := task(ctx)
            if err != nil {
                // send error if not already sent
                select {
                case errorCh <- err:
                default:
