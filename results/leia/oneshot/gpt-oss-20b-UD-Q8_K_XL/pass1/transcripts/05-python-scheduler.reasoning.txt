def best_schedule(jobs):
    n = len(jobs)
    if n == 0: return (0.0, [])
    # Build list of tuples with original index
    indexed_jobs = [(start, end, weight, idx) for idx, (start, end, weight) in enumerate(jobs)]
    # Sort by end time
    indexed_jobs.sort(key=lambda x: x[1])  # end
    ends = [job[1] for job in indexed_jobs]
    # Precompute p array
    import bisect
    p = [0]*n
    for i in range(n):
        start_i = indexed_jobs[i][0]
        # find rightmost end <= start_i
        j = bisect.bisect_right(ends, start_i) - 1
        p[i] = j  # could be -1 if none
    # DP arrays
    M = [0.0]*(n+1)  # M[0] = 0
    take = [False]*n
    for i in range(1, n+1):
        # job index i-1 in sorted list
        w = indexed_jobs[i-1][2]
        if w + M[p[i-1]+1] > M[i-1]:
            M[i] = w + M[p[i-1]+1]
            take[i-1] = True
        else:
            M[i] = M[i-1]
            take[i-1] = False
    total_weight = M[n]
    # reconstruct
    chosen = []
    i = n-1
    while i >= 0:
        if take[i]:
            chosen.append(indexed_jobs[i][3])  # original index
            i = p[i]
        else:
            i -= 1
    # chosen indices unsorted; sort by start time then index
    chosen.sort(key=lambda idx: (jobs[idx][0], idx))
    return (total_weight, chosen)
