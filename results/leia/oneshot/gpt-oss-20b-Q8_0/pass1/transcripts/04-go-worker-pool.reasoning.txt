import (
    "context"
    "errors"
    "fmt"
    "sync"
)

type Task func(ctx context.Context) (any, error)

func Run(ctx context.Context, tasks []Task, workers int) ([]any, error) {
    if workers < 1 {
        return nil, fmt.Errorf("workers must be >= 1")
    }
    if len(tasks) == 0 {
        return nil, nil
    }

    ctx2, cancel := context.WithCancel(ctx)
    defer cancel()

    results := make([]any, len(tasks))
    sem := make(chan struct{}, workers)
    var wg sync.WaitGroup
    var once sync.Once
    var firstErr error

outer:
    for i, t := range tasks {
        select {
        case <-ctx2.Done():
            break outer
        case sem <- struct{}{}:
            // proceed
        }
        wg.Add(1)
        go func(i int, t Task) {
            defer wg.Done()
            defer func() { <-sem }()
            res, err := t(ctx2)
            if err != nil {
                once.Do(func() {
                    firstErr = err
                    cancel()
                })
            }
            results[i] = res
        }(i, t)
    }

    wg.Wait()

    if firstErr != nil {
        return nil, firstErr
    }
    if ctx2.Err() != nil {
        return nil, ctx2.Err()
    }
    return results, nil
}
