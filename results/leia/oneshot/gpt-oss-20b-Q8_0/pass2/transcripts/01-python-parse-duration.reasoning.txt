import re

def parse_duration(text: str) -> float:
    if not isinstance(text, str):
        raise TypeError("text must be a string")
    s = text.strip()
    if not s:
        raise ValueError("empty duration string")
    sign = 1
    if s[0] == '-':
        sign = -1
        s = s[1:]
    if not s:
        raise ValueError("duration string missing components")
    # regex for component
    component_re = re.compile(r'(\d+(?:\.\d+)?)(h|m|s|ms)')
    pos = 0
    total = 0.0
    last_index = -1
    unit_order = {'h':0, 'm':1, 's':2, 'ms':3}
    factor = {'h':3600.0, 'm':60.0, 's':1.0, 'ms':0.001}
    while pos < len(s):
        m = component_re.match(s, pos)
        if not m:
            raise ValueError(f"invalid duration format at position {pos}")
        number_str, unit = m.group(1), m.group(2)
        # check order
        idx = unit_order[unit]
        if idx <= last_index:
            raise ValueError(f"unit order violation: {unit}")
        last_index = idx
        try:
            number = float(number_str)
        except ValueError:
            raise ValueError(f"invalid number: {number_str}")
        total += number * factor[unit]
        pos = m.end()
    if pos != len(s):
        raise ValueError("invalid trailing characters")
    return sign * total
